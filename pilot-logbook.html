Database Setup (run once in Supabase SQL Editor)

-- 1. Create tables
CREATE TABLE plb_flights (
  id TEXT PRIMARY KEY,
  username TEXT NOT NULL,
  year INT, month INT, day INT,
  aircraft_type TEXT DEFAULT '',
  civil_ident TEXT DEFAULT '',
  pic TEXT DEFAULT '',
  sic TEXT DEFAULT '',
  soc TEXT DEFAULT '',
  from_icao TEXT DEFAULT '',
  to_icao TEXT DEFAULT '',
  takeoff TEXT DEFAULT 'Day',
  landing TEXT DEFAULT 'Day',
  se NUMERIC DEFAULT 0,
  me NUMERIC DEFAULT 0,
  xc NUMERIC DEFAULT 0,
  ifr_actual NUMERIC DEFAULT 0,
  ifr_sim NUMERIC DEFAULT 0,
  notes TEXT DEFAULT '',
  updated_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE plb_duty_days (
  username TEXT,
  date_str TEXT,
  PRIMARY KEY (username, date_str)
);
CREATE TABLE plb_pilot_profiles (
  username TEXT PRIMARY KEY,
  pilot_name TEXT DEFAULT ''
);

-- 2. Enable Row Level Security + allow anon access
ALTER TABLE plb_flights ENABLE ROW LEVEL SECURITY;
ALTER TABLE plb_duty_days ENABLE ROW LEVEL SECURITY;
ALTER TABLE plb_pilot_profiles ENABLE ROW LEVEL SECURITY;
CREATE POLICY "anon_all" ON plb_flights FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "anon_all" ON plb_duty_days FOR ALL TO anon USING (true) WITH CHECK (true);
CREATE POLICY "anon_all" ON plb_pilot_profiles FOR ALL TO anon USING (true) WITH CHECK (true);

Find this in your Supabase project → SQL Editor → New query. Run it once, then come back here to connect.
